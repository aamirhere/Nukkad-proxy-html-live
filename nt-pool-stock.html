Create tables (run once in Supabase SQL editor)

CREATE TABLE IF NOT EXISTS nt_pool_stock ( product_id text PRIMARY KEY, baseline_qty int NOT NULL, baseline_at timestamptz NOT NULL, last_available int, last_sold int DEFAULT 0, updated_at timestamptz ); CREATE TABLE IF NOT EXISTS nt_pool_stock_log ( id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY, product_id text NOT NULL, sold int, delta_sold int, new_available int, ran_at timestamptz DEFAULT now() ); CREATE INDEX IF NOT EXISTS idx_pool_log_product ON nt_pool_stock_log (product_id, ran_at DESC); CREATE INDEX IF NOT EXISTS idx_pool_log_ran_at ON nt_pool_stock_log (ran_at DESC);

Reset baseline after physical recount / production top-up

UPDATE nt_pool_stock SET baseline_qty = 50, baseline_at = now(), last_available = 50, last_sold = 0, updated_at = now() WHERE product_id = 'gid://shopify/Product/123';

Daily sales summary (last 30 days)

SELECT product_id, DATE(ran_at) AS day, SUM(delta_sold) AS units_sold FROM nt_pool_stock_log WHERE ran_at >= now() - INTERVAL '30 days' GROUP BY product_id, day ORDER BY day DESC, units_sold DESC;
