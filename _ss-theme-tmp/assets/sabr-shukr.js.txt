// sabr · shukr: reveal on scroll, quick view, add to cart and the product page. Every sabr · shukr section loads
// this file; it runs once.
(function () {
  if (window.__ssInit) return;
  window.__ssInit = true;
  var ready = function (fn) { document.readyState !== 'loading' ? fn() : document.addEventListener('DOMContentLoaded', fn); };

  ready(function () {
    var io = new IntersectionObserver(function (es) { es.forEach(function (e) { if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target); } }); }, { threshold: .12 });
    document.querySelectorAll('.ss .reveal').forEach(function (el) { io.observe(el); });

    // every piece on the page, as written by snippets/ss-data.liquid
    var P = {};
    document.querySelectorAll('script[data-ss-p]').forEach(function (s) { try { var d = JSON.parse(s.textContent); P[d.handle] = d; } catch (e) {} });

    // the floating "Shop the collection" button on phones steps aside over the hero and the shop itself
    var pill = document.querySelector('.ss-pill');
    if (pill) {
      document.body.appendChild(pill);
      var seen = new Set();
      document.querySelectorAll('[data-ss-hide-pill]').forEach(function (el) {
        new IntersectionObserver(function (es) { es.forEach(function (e) { e.isIntersecting ? seen.add(e.target) : seen.delete(e.target); }); pill.classList.toggle('hide', seen.size > 0); }).observe(el);
      });
    }

    function openCart() { if (window.handleFloCartBtn) window.handleFloCartBtn(); else location.href = '/cart'; }
    function cartCount(n) {
      document.querySelectorAll('.cart-count-bubble span[aria-hidden="true"], .flo-cart-icon-bubble span[aria-hidden="true"]').forEach(function (s) { s.textContent = n; });
    }
    function addToCart(variant, qty, btns) {
      btns.forEach(function (b) { b.disabled = true; b.textContent = 'Adding\u2026'; });
      return fetch('/cart/add.js', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify({ items: [{ id: variant, quantity: qty }] }) })
        .then(function (r) { if (!r.ok) throw new Error('add'); return fetch('/cart.js'); })
        .then(function (r) { return r.json(); })
        .then(function (c) {
          cartCount(c.item_count);
          btns.forEach(function (b) { b.disabled = false; b.classList.add('added'); b.textContent = 'Added \u2713 \u00b7 view cart'; b.onclick = openCart; });
        })
        .catch(function () { btns.forEach(function (b) { b.disabled = false; b.textContent = 'Try again'; }); });
    }

    // small "Add to cart" buttons under the word pieces
    document.querySelectorAll('[data-ss-add]').forEach(function (b) {
      var d = P[b.dataset.ssAdd];
      if (!d || !d.buyable) { b.disabled = true; b.textContent = 'Coming soon'; return; }
      b.onclick = function () { addToCart(d.variant, 1, [b]); };
    });

    // quick view: a centred popup with the piece's photos, story and add to cart
    var qv = document.createElement('div');
    qv.className = 'ss ss-qv'; qv.setAttribute('role', 'dialog'); qv.setAttribute('aria-modal', 'true'); qv.setAttribute('aria-label', 'Quick view');
    qv.innerHTML = '<div class="qv-box"><button class="qv-x" type="button" aria-label="Close">\u00d7</button><div class="qv-gal"><div class="qv-thumbs"></div><div class="qv-main"></div></div><div class="qv-info"><span class="qv-tag"></span><h3></h3><span class="qv-price"></span><span class="qv-batch">Handmade to order \u00b7 7\u201310 days</span><p class="qv-q"></p><p class="qv-s"></p><ul class="qv-dl"></ul><div class="qv-buy"><div class="qty"><button type="button" data-d="-1" aria-label="Less">\u2212</button><span>1</span><button type="button" data-d="1" aria-label="More">+</button></div><button class="qv-add" type="button">Add to cart</button></div><a class="qv-more" href="#">See full details \u2192</a></div></div>';
    document.body.appendChild(qv);
    var $ = function (s) { return qv.querySelector(s); }, qty = 1;
    var esc = function (t) { return String(t == null ? '' : t).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); };
    function show(src, i) {
      $('.qv-main').innerHTML = '<img src="' + esc(src) + '" alt="">';
      qv.querySelectorAll('.qv-thumbs button').forEach(function (b, j) { b.classList.toggle('on', i === j); });
    }
    function open(h) {
      var d = P[h]; if (!d) return;
      qty = 1; $('.qty span').textContent = 1;
      $('.qv-tag').textContent = d.word ? d.word + (d.meaning ? ' \u2014 ' + d.meaning : '') : '';
      $('.qv-tag').style.display = d.word ? '' : 'none';
      $('h3').textContent = d.name; $('.qv-q').textContent = d.use || ''; $('.qv-s').textContent = d.story || '';
      $('.qv-dl').innerHTML = (Array.isArray(d.details) ? d.details : []).map(function (x) { return '<li>' + esc(x) + '</li>'; }).join('');
      $('.qv-price').textContent = d.price;
      var add = $('.qv-add'); add.classList.remove('added'); add.onclick = null;
      if (d.buyable) { add.disabled = false; add.textContent = 'Add to cart'; add.onclick = function () { addToCart(d.variant, qty, [add]); }; }
      else { add.disabled = true; add.textContent = 'Coming soon'; }
      $('.qty').style.display = d.buyable ? '' : 'none';
      $('.qv-more').href = d.url;
      var imgs = d.imgs || [];
      $('.qv-gal').classList.toggle('single', imgs.length < 2);
      $('.qv-thumbs').innerHTML = imgs.map(function (s) { return '<button type="button" aria-label="Photo"><img src="' + esc(s) + '" alt=""></button>'; }).join('');
      qv.querySelectorAll('.qv-thumbs button').forEach(function (b, i) { b.onclick = function () { show(imgs[i], i); }; });
      if (imgs.length) show(imgs[0], 0); else $('.qv-main').innerHTML = '<span class="ss-word-tile">' + esc(d.word || d.name) + '</span>';
      qv.classList.add('open'); document.documentElement.style.overflow = 'hidden'; $('.qv-x').focus();
    }
    function close() { qv.classList.remove('open'); $('.qv-main').innerHTML = ''; document.documentElement.style.overflow = ''; }
    $('.qv-x').onclick = close;
    qv.addEventListener('click', function (e) { if (e.target === qv) close(); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape' && qv.classList.contains('open')) close(); });
    qv.querySelectorAll('.qty button').forEach(function (b) { b.onclick = function () { qty = Math.max(1, qty + (+b.dataset.d)); $('.qty span').textContent = qty; }; });
    document.addEventListener('click', function (e) {
      var t = e.target.closest('[data-ss-qv]');
      if (!t || qv.contains(t) || !P[t.dataset.ssQv]) return;
      e.preventDefault(); open(t.dataset.ssQv);
    });

    // product page: gallery, quantity, add to cart and the buy bar that follows on phones
    var pd = document.querySelector('.ss .pd[data-ss-handle]');
    if (pd) {
      var d = P[pd.dataset.ssHandle] || {}, pq = 1;
      var thumbs = pd.querySelectorAll('.thumbs button'), main = pd.querySelector('.gal .main');
      thumbs.forEach(function (b, i) { b.onclick = function () { main.innerHTML = '<img src="' + esc(b.dataset.src) + '" alt="">'; thumbs.forEach(function (x, j) { x.classList.toggle('on', i === j); }); }; });
      pd.querySelectorAll('.buy .qty button').forEach(function (b) { b.onclick = function () { pq = Math.max(1, pq + (+b.dataset.d)); pd.querySelector('.buy .qty span').textContent = pq; }; });
      var addBtn = pd.querySelector('[data-ss-buy]'), bar = document.querySelector('.ss-bar-buy'), barBtn = bar && bar.querySelector('button');
      var btns = [addBtn, barBtn].filter(Boolean);
      if (!d.buyable) btns.forEach(function (b) { b.disabled = true; b.textContent = 'Coming soon'; });
      else {
        addBtn.onclick = function () { addToCart(d.variant, pq, btns); };
        if (barBtn) barBtn.onclick = function () { addToCart(d.variant, 1, btns); };
      }
      if (bar && addBtn) {
        document.body.appendChild(bar);
        new IntersectionObserver(function (es) { var e = es[0]; bar.classList.toggle('show', !e.isIntersecting && e.boundingClientRect.top < 0); }).observe(addBtn);
      }
    }
  });
})();
